{#
  Dimension minimale : enrichie par le référentiel articles (stock*.xlsx)
  si disponible, sinon colonnes dérivées des transactions.
#}

with transactions as (

    select * from {{ ref('stg_erp_ventes_achats') }}

),

par_article as (

    select
        numero_article,
        min(date_ecriture)                                     as premiere_transaction,
        max(date_ecriture)                                     as derniere_transaction,
        count(*)                                                as nb_lignes,
        count(*) filter (where type_ecriture = 'Vente')         as nb_lignes_vente,
        count(*) filter (where type_ecriture = 'Achat')         as nb_lignes_achat,
        count(*) filter (where is_retour_vente)                 as nb_lignes_retour,
        count(distinct code_unite)                              as nb_unites_distinctes,
        mode() within group (order by code_unite)               as unite_principale

    from transactions
    group by numero_article

),

-- Référentiel articles (si la table existe)
ref_articles as (

    select * from {{ ref('stg_erp_referentiel_articles') }}

)

select
    {{ dbt_utils.generate_surrogate_key(['p.numero_article']) }}  as article_key,
    p.numero_article,
    r.description                                                 as description_article,
    r.stock                                                        as stock_initial,
    r.unite_base                                                   as unite_base_referentiel,
    p.premiere_transaction,
    p.derniere_transaction,
    p.nb_lignes,
    p.nb_lignes_vente,
    p.nb_lignes_achat,
    p.nb_lignes_retour,
    p.unite_principale,
    p.nb_unites_distinctes > 1                                     as a_plusieurs_unites

from par_article p
left join ref_articles r on p.numero_article = r.numero_article
